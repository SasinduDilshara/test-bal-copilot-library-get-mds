## Module: aws.s3

## Overview
The Ballerina AWS S3 provides the capability to manage buckets and objects in [AWS S3](https://aws.amazon.com/s3/).

This module supports [Amazon S3 REST API](https://docs.aws.amazon.com/AmazonS3/latest/API/Welcome.html) `2006-03-01` version.
 
## Prerequisites
Before using this connector in your Ballerina application, complete the following:
1. Create an [AWS account](https://portal.aws.amazon.com/billing/signup?nc2=h_ct&src=default&redirect_url=https%3A%2F%2Faws.amazon.com%2Fregistration-confirmation#/start)
2. [Obtain tokens](https://docs.aws.amazon.com/IAM/latest/UserGuide/id_credentials_access-keys.html)

## Quickstart
To use the AWS S3 connector in your Ballerina application, update the .bal file as follows:

### Step 1: Import connector
Import the `ballerinax/aws.s3` module into the Ballerina project.
```ballerina
import ballerinax/aws.s3;
```

### Step 2: Create a new connector instance
Create a `s3:ConnectionConfig` with the tokens obtained, and initialize the connector with it.
```ballerina
s3:ConnectionConfig amazonS3Config = {
    accessKeyId: <ACCESS_KEY_ID>,
    secretAccessKey: <SECRET_ACCESS_KEY>,
    region: <REGION>
};

s3:Client amazonS3Client = check new(amazonS3Config);
```

IAM role-based authentication can be used as below if the code is running within an EC2 instance.
```ballerina
s3:ConnectionConfig amazonS3Config = {
    authType: s3:EC2_IAM_ROLE,
    region: <REGION>
} ;
```

### Step 3: Invoke connector operation
1. Now you can use the operations available within the connector. Note that they are in the form of remote operations.  
Following is an example on how to create a bucket using the connector.

    ```ballerina
    string bucketName = "name";

    public function main() returns error? {
        _ = check amazonS3Client->createBucket(bucketName);
    }
    ```
2. Use `bal run` command to compile and run the Ballerina program.

**[You can find a list of samples here](https://github.com/ballerina-platform/module-ballerinax-aws.s3/tree/master/examples)**

# Ballerina Amazon S3 connector provides the capability to access AWS S3 API.
# This connector lets you to get authorized access to AWS S3 buckets and objects.
# 
client class Client {
    function init(ConnectionConfig config) returns error?;

    # Retrieves a list of all Amazon S3 buckets that the authenticated user of the request owns.
    # 
    remote function listBuckets() returns Bucket[]|error;

    # Creates a bucket.
    # 
    remote function createBucket(string bucketName, "private"|"public-read"|"public-read-write"|"aws-exec-read"|"authenticated-read"|"bucket-owner-read"|"bucket-owner-full-control"|() cannedACL = ()) returns error?;

    # Retrieves the existing objects in a given bucket.
    # 
    remote function listObjects(string bucketName, string|() delimiter = (), string|() encodingType = (), int|() maxKeys = (), string|() prefix = (), string|() startAfter = (), boolean|() fetchOwner = (), string|() continuationToken = ()) returns S3Object[]|error;

    # Retrieves objects from Amazon S3.
    # 
    remote function getObject(string bucketName, string objectName, ObjectRetrievalHeaders|() objectRetrievalHeaders = (), int|() byteArraySize = ()) returns Error?>|error;

    # Creates an object.
    # 
    remote function createObject(string bucketName, string objectName, string|xml|json|byte[]|stream<io:Block, io:Error?> payload, "private"|"public-read"|"public-read-write"|"aws-exec-read"|"authenticated-read"|"bucket-owner-read"|"bucket-owner-full-control"|() cannedACL = (), ObjectCreationHeaders|() objectCreationHeaders = (), map<string> userMetadataHeaders = {}) returns error?;

    # Deletes an object.
    # 
    remote function deleteObject(string bucketName, string objectName, string|() versionId = ()) returns error?;

    # Deletes a bucket.
    # 
    remote function deleteBucket(string bucketName) returns error?;

    # Generates a presigned URL for the object.
    # 
    remote function createPresignedUrl(string bucketName, string objectName, "RETRIEVE"|"CREATE"|ObjectCreationHeaders|ObjectRetrievalHeaders action, int expires = 0, int|() partNo = (), string|() uploadId = ()) returns string|error;

    # Initiates a multipart upload and returns an upload ID.
    # 
    remote function createMultipartUpload(string objectName, string bucketName, "private"|"public-read"|"public-read-write"|"aws-exec-read"|"authenticated-read"|"bucket-owner-read"|"bucket-owner-full-control"|() cannedACL = (), MultipartUploadHeaders|() multipartUploadHeaders = ()) returns string|error;

    # Completes a multipart upload by assembling previously uploaded parts.
    # 
    remote function uploadPart(string objectName, string bucketName, string|xml|json|byte[]|stream<io:Block, io:Error?> payload, string uploadId, int partNumber, UploadPartHeaders|() uploadPartHeaders = ()) returns CompletedPart|error;

    # Completes a multipart upload by assembling previously uploaded parts.
    # 
    remote function completeMultipartUpload(string objectName, string bucketName, string uploadId, CompletedPart[] completedParts) returns error?;

    # Aborts a multipart upload.
    # 
    remote function abortMultipartUpload(string objectName, string bucketName, string uploadId) returns error?;
}
