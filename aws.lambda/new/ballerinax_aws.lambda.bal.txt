// --- Types ---

class Context {
    function init(string requestId, int deadlineMs, string invokedFunctionArn, string traceId) returns ();

    # Returns the unique id for this request.
    function getRequestId() returns string;

    # Returns the request execution deadline in milliseconds from the epoch.
    function getDeadlineMs() returns int;

    # Returns the ARN of the function being invoked.
    function getInvokedFunctionArn() returns string;

    # Returns the trace id for this request
    function getTraceId() returns string;

    # Returns the remaining execution time for this request in milliseconds
    function getRemainingExecutionTime() returns int;
}

# Represents the details of the identity related to the S3 service.
# 

type S3Identity record {
    # S3 principal ID
    string principalId;
};

# Represents the details of the bucket related to the S3 service.
# 

type S3Bucket record {
    # S3 bucket name
    string name;
    # S3 bucket owners identity
    S3Identity ownerIdentity;
    # S3 bucket arn
    string arn;
};

# Represents the details of the object related to the S3 service.
# 

type S3Object record {
    # Object key
    string key;
    # Object size
    int size;
    # Object tag
    string eTag;
    # Object sequencer  
    string sequencer;
};

# Represents the details of the element related to the S3 service.
# 

type S3Element record {
    # Version of the S3 schema
    string s3SchemaVersion;
    # Configuration ID
    string configurationId;
    # S3 bucket related details
    S3Bucket bucket;
    S3Object 'object;
};

# Represents the S3 bucket notification related details.
# 

type S3Record record {
    # version of the event
    string eventVersion;
    # the source of the triggered event
    string eventSource;
    # the AWS region to which the S3 bucket belongs 
    string awsRegion;
    # invoked time of the event
    string eventTime;
    # name of the invoked event
    string eventName;
    # identity of the user who invoked the event
    S3Identity userIdentity;
    # parameters of the request
    map<string> requestParameters;
    # the response elements
    map<string> responseElements;
    # S3 element related details
    S3Element s3;
};

# Represents the S3 event details received from AWS when the S3 notification is triggered.
# 

type S3Event record {
    # A list of S3 event notification records
    S3Record[] Records;
};

# Represents the Amazon simple queue service notification related details.
# 

type SQSRecord record {
    # ID of the message
    string messageId;
    # receipt handle of the message 
    string receiptHandle;
    # body of the notification
    string body;
    # the attributes associated with the queue
    map<string> attributes;
    # attributes of the message
    map<string> messageAttributes;
    # md5 hash of the body
    string md5OfBody;
    # source of the triggered event
    string eventSource;
    # arn of the event source
    string eventSourceARN;
    # region of the SQS notification
    string awsRegion;
};

# Represents the AWS SQS event details received from AWS when the SQS notification is triggered.
# 

type SQSEvent record {
    # list of SQS event notifications
    SQSRecord[] Records;
};

# Represents the AWS API Gateway proxy request details received from AWS when the gateway is triggered.
# 

type APIGatewayProxyRequest record {
    string 'resource;
    # the URL path of the caller
    string path;
    # HTTP method of the request
    string httpMethod;
    # request context of the request
    map<string> requestContext;
    # headers of the request
    map<string> headers;
    # multi value headers (if they are enabled)
    map<string[]> multiValueHeaders;
    # query string parameters of the request
    map<string>? queryStringParameters;
    # multi value query string parameters of the request
    map<string[]>? multiValueQueryStringParameters;
    # path parameters of the request
    map<string>? pathParameters;
    # stage variables  of the stage in the API Gateway
    map<string>? stageVariables;
    # body of the request
    string? body;
    # field to identify if the content is Base64 encoded
    boolean isBase64Encoded;
};

# Represents the details of the stream related to the DynamoDB.
# 

type DynamoDBStreamRecord record {
    # the primary key attribute(s) of the DynamoDB item that were modified
    map<json> Keys;
    # the item in the DynamoDB table after it is modified
    map<json> NewImage?;
    # the item in the DynamoDB table before it was modified
    map<json> OldImage?;
    # the type of data of the modified DynamoDB item that was captured in this stream record
    string StreamViewType;
    # the sequence number of the stream record
    string SequenceNumber;
    # the size of the stream record (in bytes)
    int SizeBytes;
};

# Represents the DynamoDB related details.
# 

type DynamoDBRecord record {
    # a globally unique identifier for the event that was recorded in this stream record
    string eventID;
    # the version number of the stream record format
    string eventVersion;
    # the main body of the stream record containing all of the DynamoDB-specific fields
    DynamoDBStreamRecord dynamodb;
    # AWS region to which the DynamoDB belongs
    string awsRegion;
    # the type of data modification that was performed on the DynamoDB table
    string eventName;
    # the event source arn of then DynamoDB
    string eventSourceARN;
    # the AWS service from which the stream record originated
    string eventSource;
};

# Represents the DynamoDB event details received from AWS when the DynamoDB notification is triggered.
# 

type DynamoDBEvent record {
    # record list of the DynamoDB notifications
    DynamoDBRecord[] Records;
};

# Represents the simple email service related details.
# 

type SESCommonHeaders record {
    string[] 'from;
    # the receiver address of the email
    string[] to;
    # return path of the email
    string returnPath;
    # the unique ID assigned to the email by Amazon SES
    string messageId;
    # the day on which the email was sent
    string date;
    # subject of the email
    string subject;
};

# Represents the details of the name related to the simple email service.
# 

type NameValue record {
    # name of the header
    string name;
    # value of the header
    string value;
};

# Represents the details of the email related to the simple email service.
# 

type SESMail record {
    # list of headers common to all emails
    SESCommonHeaders commonHeaders;
    string 'source;
    # string that specifies the date and time at which the action was triggered (in the ISO 8601 format)
    string timestamp;
    # a complete list of all recipient addresses
    string[] destination;
    # parameter description  
    NameValue[] headers;
    # specifies whether the headers were truncated in the notification
    boolean headersTruncated;
    # the unique ID assigned to the email by Amazon SES
    string messageId;
};

# Represents the details of the verdict related to the simple email service.
# 

type SESVerdict record {
    # status of the verdict
    string status;
};

# Represents the details of the action related to the simple email service.
# 

type SESAction record {
    string 'type;
    # invocation type of the email service
    string invocationType;
    # arn of the function that is executed
    string functionArn;
};

# Represents the details of the receipt related to the simple email service.
# 

type SESReceipt record {
    # a list of recipients that were matched by the active receipt rule
    string[] recipients;
    # string that specifies the date and time at which the action was triggered (in the ISO 8601 format)
    string timestamp;
    # object that indicates whether the message is spam
    SESVerdict spamVerdict;
    # object that indicates whether the DomainKeys Identified Mail (DKIM) check passed
    SESVerdict dkimVerdict;
    # processing time taken for the email (in milliseconds)
    int processingTimeMillis;
    # Object that encapsulates the information about the action that was executed
    SESAction action;
    # object that indicates whether the Sender Policy Framework (SPF) check passed
    SESVerdict spfVerdict;
    # virus verdict status of the email
    SESVerdict virusVerdict;
};

# Represents the details of the element related to the simple email service.
# 

type SESElement record {
    # email-related details of the event
    SESMail mail;
    # receipt-related details of the event
    SESReceipt receipt;
};

# Represents the details of the notification related to the simple email service.
# 

type SESRecord record {
    # the version number of the email format
    string eventVersion;
    # details of the SES event
    SESElement ses;
    # the AWS service from which the email originated
    string eventSource;
};

# Represents the simple email service event details received from AWS when the SES notification is triggered.
# 

type SESEvent record {
    # record list of the SES notifications
    SESRecord[] Records;
};

# Register a function handler with the function and event type.
# 
# + handler - Function Hanlder name
# + func - Function type
# + eventType - Event type
function __register(string handler, FunctionType func, anydata eventType) returns ();

# Process and excute the handler.  
function __process() returns ();

// --- Annotations ---

# The annotation, which is used to mark the function as an AWS Lambda function.
public annotation Function on function;
