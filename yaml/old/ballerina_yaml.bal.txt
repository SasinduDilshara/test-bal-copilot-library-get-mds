import ballerina/yaml;

// --- README ---
## Overview

This module provides APIs to convert a YAML configuration file to json, and vice-versa.

Since the parser is following LL(1) grammar, it follows a non-recursive predictive parsing algorithm which operates in a linear time complexity.

For information on the operations, which you can perform with the `yaml` module, see the below **Functions**.

# Configurations for writing a YAML document.
# 

type WriteConfig record {
    # Number of whitespace for an indentation  
    int indentationPolicy?;
    # The maximum depth level for a block collection 
    int blockLevel?;
    # If set, the tags are written along with the nodes
    boolean canonical?;
    # If set, single quotes are used to surround scalars
    boolean useSingleQuotes?;
    # If set, all the scalars are surrounded by quotes
    boolean forceQuotes?;
    # YAML schema used for writing
    YAMLSchema schema?;
    # Custom YAML types for the schema
    YamlType[] yamlTypes?;
    # Custom tag handles that can be included as directives
    map<string> customTagHandles?;
    # If set, the parser will write a stream of YAML documents
    boolean isStream?;
};

# Represents the YAML schema available for the parser.
# 
enum YAMLSchema {
    CORE_SCHEMA,
    JSON_SCHEMA,
    FAILSAFE_SCHEMA
}

# Represents the attributes of the custom YAML type.
# 

type YamlType record {
    # YAML tag for the custom type  
    string tag;
    # The equivalent Ballerina type for the YAML tag  
    typedesc<json> ballerinaType;
    # Fail safe schema type
    FailSafeSchema kind;
    # Function to generate the Ballerina data structure.  
    function (json data) returns json|ballerina/yaml:0.8.0:SchemaError construct;
    # Function to convert the Ballerina data structure to YAML.
    function (json data) returns string|ballerina/yaml:0.8.0:SchemaError represent;
};

# Represents the basic YAML types available in the Fail safe schema.
# 
enum FailSafeSchema {
    STRING,
    SEQUENCE,
    MAPPING
}

# Configurations for reading a YAML document.
# 

type ReadConfig record {
    # YAML schema used for writing  
    YAMLSchema schema?;
    # Custom YAML types for the schema  
    YamlType[] yamlTypes?;
    # If set, the parser reads a stream of YAML documents  
    boolean isStream?;
    # Flag is set if anchors can be redefined multiple times
    boolean allowAnchorRedefinition?;
    # Flag is set if same map keys are allowed in a mapping
    boolean allowMapEntryRedefinition?;
};
