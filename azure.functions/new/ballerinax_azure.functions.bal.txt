## Package Overview

This module provides an annotation-based [Azure Functions](https://azure.microsoft.com/en-us/services/functions/) extension implementation for Ballerina. 

- For information on the operations, which you can perform with this module, see [API Docs](https://docs.central.ballerina.io/ballerinax/azure-functions/). 
- For more information on the deployment, see the [Azure Functions Deployment Guide](https://ballerina.io/swan-lake/learn/user-guide/deployment/azure-functions/).
- For examples on the usage of the operations, see the [Azure Functions Deployment Example](https://ballerina.io/swan-lake/learn/by-example/azure-functions-deployment.html).

### Azure Setup

* An Azure "Function App" needs to be created in a given resource group with the following requirements
   - Runtime stack - "Java 21"
   - Hosting operating system - "Windows" (This is the default and Linux is not supported in Azure for custom handlers at the moment)

## Supported Triggers and bindings:

- Http - Trigger and Output Binding
- Queue - Trigger and Output Binding
- Blob - Trigger, Input and Output Binding
- Twilio - Output Binding
- CosmosDB - Trigger, Input and Output Binding
- Timer - Trigger

// --- Types ---


type FunctionConfiguration record {
    string name;
};

type AUTH_LEVEL "anonymous"|"function"|"admin";

# HTTPTrigger annotation configuration.
# 

type HTTPTriggerConfiguration record {
    # The authentication level of the function
    AUTH_LEVEL authLevel?;
};

# Defines the Header resource signature parameter.
# 

type HttpHeader record {
    # Specifies the name of the required header
    string name?;
};

# Queue annotation configuration.
# 

type QueueConfiguration record {
    # The queue name
    string queueName;
    # The name of the app setting which contains the Storage connection string
    string connection?;
};

# TimerTrigger annotation configuration.
# 

type TimerTriggerConfiguration record {
    # The CRON expression representing the timer schedule.
    string schedule;
    # The flag to state if the timer should be started on a runtime restart
    boolean runOnStartup?;
};

# Blob annotation configuration.
# 

type BlobConfiguration record {
    # The blob container path
    string path;
    # The name of the app setting which contains the Storage connection string
    string connection?;
};

# CosmosDB trigger annotation configuration.
# 

type CosmosDBTriggerConfiguration record {
    # The name of the app setting which contains the connection string for CosmosDB account
    string connectionStringSetting;
    # The database name
    string databaseName;
    # The collection name
    string collectionName;
    # The lease collection is automatically created when this is set to true
    boolean createLeaseCollectionIfNotExists?;
    # The request throughput of the lease collection
    int leasesCollectionThroughput?;
};

# CosmosDB input annotation configuration.
# 

type CosmosDBInputConfiguration record {
    # The name of the app setting which contains the connection string for CosmosDB account
    string connectionStringSetting;
    # The database name
    string databaseName;
    # The collection name
    string collectionName;
    # The id of the document to retrieve
    string id?;
    # An Azure Cosmos DB SQL query used to retrieve multiple documents
    string sqlQuery?;
    # The partition key value for lookups
    string|int|float partitionKey?;
};

# CosmosDB output annotation configuration.
# 

type CosmosDBOutputConfiguration record {
    # The name of the app setting which contains the connection string for CosmosDB account
    string connectionStringSetting;
    # The database name
    string databaseName;
    # The collection name
    string collectionName;
};

# Twilio annotation configuration.
# 

type TwilioSmsConfiguration record {
    # The app setting which holds the Twilio Account Sid
    string accountSidSetting?;
    # The app setting which holds the Twilio authentication token
    string authTokenSetting?;
    string 'from;
    # The phone number the SMS is sent to
    string to;
};

# BindingName annotation configuration.
# 

type BindingNameConfiguration record {
    # The binding name
    string name?;
};

# 

type TimerMetadata record {
    # Schedule the timer is being executed on  
    TimerSchedule Schedule;

#

};

type TwilioSmsOutputBinding record {

    string to?;
    # The message body
    string body?;
};
